i,j,aux, trocou: inteiro
inicio

         trocou <- 0 // a cada nova iteração a variável trocou recebe 0.
         para j de 0 ate 8 faca //até 8 para que a posição v[8] possa comparar com a v[j+1] = v[9].
                  se (v[j] > v[j+1]) entao
                         aux<- v[j]
                         v[j] <- v[j+1]
                         v[j+1] <- aux
                         trocou <- 1 // se houve troca a váriavel recebe 1.
                  fimse
         fimpara
         
         escreval(i, "ª iteração")
         i <- i+1
